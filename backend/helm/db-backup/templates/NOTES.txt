1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "db-backup.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "db-backup.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "db-backup.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "db-backup.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

2. Check the application health:
  kubectl --namespace {{ .Release.Namespace }} get pods -l "app.kubernetes.io/name={{ include "db-backup.name" . }},app.kubernetes.io/instance={{ .Release.Name }}"

3. View the logs:
  kubectl --namespace {{ .Release.Namespace }} logs -l "app.kubernetes.io/name={{ include "db-backup.name" . }},app.kubernetes.io/instance={{ .Release.Name }}"

4. Access metrics (if Prometheus is enabled):
  {{- if .Values.ingress.enabled }}
  curl http{{ if .Values.ingress.tls }}s{{ end }}://{{ (index .Values.ingress.hosts 0).host }}/metrics
  {{- else }}
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "db-backup.fullname" . }} 9090:9090
  curl http://localhost:9090/metrics
  {{- end }}

{{- if .Values.autoscaling.enabled }}

5. HorizontalPodAutoscaler is enabled:
  kubectl --namespace {{ .Release.Namespace }} get hpa {{ include "db-backup.fullname" . }}
{{- end }}

{{- if .Values.persistence.enabled }}

6. Persistent storage is enabled:
  kubectl --namespace {{ .Release.Namespace }} get pvc {{ include "db-backup.pvcName" . }}
{{- end }}

{{- if .Values.istio.enabled }}

7. Istio service mesh integration is enabled:
  - Virtual Service: kubectl --namespace {{ .Release.Namespace }} get virtualservice {{ include "db-backup.fullname" . }}
  - Destination Rule: kubectl --namespace {{ .Release.Namespace }} get destinationrule {{ include "db-backup.fullname" . }}
  - Gateway: kubectl --namespace {{ .Release.Namespace }} get gateway {{ include "db-backup.fullname" . }}-gateway
{{- end }}
